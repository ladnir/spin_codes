cmake_minimum_required(VERSION 3.20)
project(spin VERSION 0.4.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(SPIN_ENABLE_AVX512 "Build runtime-dispatched AVX-512 kernels" ON)
set(SPIN_TESTS_DEFAULT OFF)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  set(SPIN_TESTS_DEFAULT ON)
endif()
option(SPIN_BUILD_TESTS "Build standalone API tests" ${SPIN_TESTS_DEFAULT})
option(SPIN_BUILD_BENCHMARKS "Build serial setup/encoding benchmarks" OFF)
set(SPIN_TUNE generic CACHE STRING "Compiler scheduling target (not required ISA)")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "SPIN requires a 64-bit target")
endif()
# Turbo SPIN (PacketRs8T64) is portable C++ with runtime-dispatched ISA kernels:
# AVX-512 and AVX2(+GFNI) on x86-64, NEON on AArch64, portable elsewhere.
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
  set(SPIN_X86 ON)
else()
  set(SPIN_X86 OFF)
  set(SPIN_ENABLE_AVX512 OFF CACHE BOOL "Build runtime-dispatched AVX-512 kernels" FORCE)
endif()
add_library(spin STATIC src/Code.cpp src/PreparedEncoder.cpp src/Buffer.cpp src/Cpu.cpp
  src/rs8/SetupScalar.cpp src/rs8/Portable.cpp src/rs8/Neon.cpp
  src/certified22/Setup.cpp src/certified22/SetupPortable.cpp src/certified22/Scalar.cpp)
add_library(spin::spin ALIAS spin)
target_compile_features(spin PUBLIC cxx_std_20)
target_include_directories(spin PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_definitions(spin PRIVATE SPIN_AVX512=$<BOOL:${SPIN_ENABLE_AVX512}> SPIN_WORKSPACE_ROUTING_OPT=1)
set_target_properties(spin PROPERTIES POSITION_INDEPENDENT_CODE ON INTERPROCEDURAL_OPTIMIZATION FALSE)
if(NOT MSVC)
  target_compile_options(spin PRIVATE -fno-strict-aliasing -fno-lto)
endif()
# AVX2 kernels, runtime-dispatched: AVX2 only, plus a GFNI variant.
if(SPIN_X86)
  add_library(spin_turbo256 OBJECT src/rs8/Avx2.cpp src/rs8/Avx2Gfni.cpp src/certified22/SetupAvx2.cpp)
  if(NOT MSVC)
    set_source_files_properties(src/rs8/Avx2Gfni.cpp PROPERTIES COMPILE_OPTIONS "-mgfni")
  endif()
  target_compile_features(spin_turbo256 PRIVATE cxx_std_20)
  set_target_properties(spin_turbo256 PROPERTIES POSITION_INDEPENDENT_CODE ON INTERPROCEDURAL_OPTIMIZATION FALSE)
  if(MSVC)
    target_compile_options(spin_turbo256 PRIVATE /arch:AVX2)
  else()
    target_compile_options(spin_turbo256 PRIVATE -mavx2 -mtune=${SPIN_TUNE} -fno-strict-aliasing -fno-lto)
  endif()
  target_sources(spin PRIVATE $<TARGET_OBJECTS:spin_turbo256>)
endif()
if(SPIN_ENABLE_AVX512)
  # AVX-512 ISA stays in private translation units. Neither setup dispatch nor
  # the public API is compiled for AVX-512.
  add_library(spin_turbo512 OBJECT src/rs8/Fast.cpp src/rs8/ForwardFast.cpp
    src/certified22/RouteFast.cpp src/certified22/SetupFast.cpp)
  target_compile_features(spin_turbo512 PRIVATE cxx_std_20)
  target_compile_definitions(spin_turbo512 PRIVATE SPIN_AVX512=1)
  set_target_properties(spin_turbo512 PROPERTIES POSITION_INDEPENDENT_CODE ON INTERPROCEDURAL_OPTIMIZATION FALSE)
  if(MSVC)
    target_compile_options(spin_turbo512 PRIVATE /arch:AVX512)
    target_compile_definitions(spin_turbo512 PRIVATE SPIN_PACKET_ENABLE_VBMI=1)
  else()
    target_compile_options(spin_turbo512 PRIVATE -mavx2 -mavx512f -mavx512vl -mavx512bw -mavx512dq
      -mavx512vbmi -mgfni -mtune=${SPIN_TUNE} -fno-strict-aliasing -fno-lto)
  endif()
  target_sources(spin PRIVATE $<TARGET_OBJECTS:spin_turbo512>)
endif()
install(TARGETS spin EXPORT spinTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/spin)
install(EXPORT spinTargets NAMESPACE spin:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/spin)
export(EXPORT spinTargets NAMESPACE spin:: FILE ${CMAKE_CURRENT_BINARY_DIR}/spinTargets.cmake)
configure_package_config_file(cmake/spinConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/spinConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/spin)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/spinConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/spinConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/spinConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/spin)
if(SPIN_BUILD_BENCHMARKS)
  add_executable(spin_turbo_backends_bench bench/turbo_backends.cpp)
  target_link_libraries(spin_turbo_backends_bench PRIVATE spin::spin)
  add_executable(spin_setup_once_bench bench/setup_once.cpp)
  target_link_libraries(spin_setup_once_bench PRIVATE spin::spin)
endif()
if(SPIN_BUILD_BENCHMARKS AND SPIN_ENABLE_AVX512)
  # Stage timing calls internal AVX2 and AVX-512 entry points directly.
  add_executable(spin_rs8_stages_bench bench/rs8_stages.cpp)
  target_link_libraries(spin_rs8_stages_bench PRIVATE spin::spin)
endif()
if(SPIN_BUILD_TESTS)
  enable_testing()
  add_executable(spin_rs8_test tests/rs8.cpp)
  target_link_libraries(spin_rs8_test PRIVATE spin::spin)
  add_test(NAME spin_rs8 COMMAND spin_rs8_test)
  add_test(NAME spin_rs8_avx2 COMMAND spin_rs8_test avx2)
  add_test(NAME spin_rs8_gfni COMMAND spin_rs8_test gfni)
  add_test(NAME spin_rs8_portable COMMAND spin_rs8_test portable)
  add_test(NAME spin_rs8_neon COMMAND spin_rs8_test neon)
  set_tests_properties(spin_rs8 spin_rs8_portable PROPERTIES RUN_SERIAL TRUE)
  set_tests_properties(spin_rs8_avx2 spin_rs8_gfni spin_rs8_neon PROPERTIES RUN_SERIAL TRUE SKIP_RETURN_CODE 77)
  add_executable(spin_turbo_setup_test tests/turbo_setup.cpp)
  target_link_libraries(spin_turbo_setup_test PRIVATE spin::spin)
  add_test(NAME spin_turbo_setup COMMAND spin_turbo_setup_test)
  set_tests_properties(spin_turbo_setup PROPERTIES RUN_SERIAL TRUE SKIP_RETURN_CODE 77)
  add_executable(spin_setup_test tests/setup.cpp)
  target_compile_features(spin_setup_test PRIVATE cxx_std_20)
  add_test(NAME spin_setup COMMAND spin_setup_test)
endif()
