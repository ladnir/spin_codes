cmake_minimum_required(VERSION 3.20)
project(spin VERSION 0.2.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(SPIN_ENABLE_AVX512 "Build runtime-dispatched AVX-512 kernels" ON)
set(SPIN_TESTS_DEFAULT OFF)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  set(SPIN_TESTS_DEFAULT ON)
endif()
option(SPIN_BUILD_TESTS "Build standalone API tests" ${SPIN_TESTS_DEFAULT})
option(SPIN_BUILD_BENCHMARKS "Build serial setup/encoding benchmark" OFF)
option(SPIN_BUILD_EXPERIMENTS "Build uncertified permutation routing experiments" OFF)
set(SPIN_TUNE generic CACHE STRING "Compiler scheduling target (not required ISA)")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
  message(FATAL_ERROR "This SPIN release requires 64-bit x86; ARM is not yet supported")
endif()
set(KERNEL src/kernels)
add_library(spin_kernels OBJECT ${KERNEL}/Spin.cpp ${KERNEL}/BankKernel.cpp ${KERNEL}/generated/BchCircuit.cpp
  ${KERNEL}/generated/BchForward.cpp ${KERNEL}/wide/Wide256.cpp)
target_include_directories(spin_kernels PRIVATE ${KERNEL} include)
target_compile_features(spin_kernels PRIVATE cxx_std_20)
target_compile_definitions(spin_kernels PRIVATE SPIN_BCH_AVX512=$<BOOL:${SPIN_ENABLE_AVX512}>
  SPIN_GROUPED_A=1 SPIN_WORKSPACE_ROUTING_OPT=1)
set_target_properties(spin_kernels PROPERTIES POSITION_INDEPENDENT_CODE ON INTERPROCEDURAL_OPTIMIZATION FALSE)
if(MSVC)
  target_compile_options(spin_kernels PRIVATE /arch:AVX2)
else()
  target_compile_options(spin_kernels PRIVATE -mavx2 -mtune=${SPIN_TUNE} -fno-strict-aliasing -fno-lto)
endif()
add_library(spin STATIC src/Code.cpp src/PreparedEncoder.cpp src/Cpu.cpp $<TARGET_OBJECTS:spin_kernels>)
add_library(spin::spin ALIAS spin)
target_compile_features(spin PUBLIC cxx_std_20)
target_include_directories(spin PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_definitions(spin PRIVATE SPIN_BCH_AVX512=$<BOOL:${SPIN_ENABLE_AVX512}> SPIN_WORKSPACE_ROUTING_OPT=1)
set_target_properties(spin PROPERTIES POSITION_INDEPENDENT_CODE ON INTERPROCEDURAL_OPTIMIZATION FALSE)
if(NOT MSVC)
  target_compile_options(spin PRIVATE -fno-strict-aliasing -fno-lto)
endif()
if(SPIN_ENABLE_AVX512)
  add_library(spin_kernels512 OBJECT ${KERNEL}/Fast.cpp ${KERNEL}/ForwardFast.cpp ${KERNEL}/BankFast.cpp ${KERNEL}/BankMasks.cpp
    ${KERNEL}/generated/BchAvx512.cpp ${KERNEL}/generated/BchForward512.cpp ${KERNEL}/wide/Wide512.cpp)
  target_compile_features(spin_kernels512 PRIVATE cxx_std_20)
  target_include_directories(spin_kernels512 PRIVATE ${KERNEL} include)
  target_compile_definitions(spin_kernels512 PRIVATE SPIN_GROUPED_A=1 SPIN_BCH_AVX512=1)
  set_target_properties(spin_kernels512 PROPERTIES POSITION_INDEPENDENT_CODE ON INTERPROCEDURAL_OPTIMIZATION FALSE)
  if(MSVC)
    target_compile_options(spin_kernels512 PRIVATE /arch:AVX512)
  else()
    target_compile_options(spin_kernels512 PRIVATE -mavx2 -mavx512f -mavx512vl -mtune=${SPIN_TUNE} -fno-strict-aliasing -fno-lto)
    set_source_files_properties(${KERNEL}/wide/Wide512.cpp PROPERTIES COMPILE_OPTIONS "-mavx512bw;-mavx512dq")
    set_source_files_properties(${KERNEL}/BankMasks.cpp PROPERTIES COMPILE_OPTIONS "-mavx512bw;-mavx512dq;-mavx512vpopcntdq")
  endif()
  target_sources(spin PRIVATE $<TARGET_OBJECTS:spin_kernels512>)
endif()
install(TARGETS spin EXPORT spinTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/spin)
install(EXPORT spinTargets NAMESPACE spin:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/spin)
export(EXPORT spinTargets NAMESPACE spin:: FILE ${CMAKE_CURRENT_BINARY_DIR}/spinTargets.cmake)
configure_package_config_file(cmake/spinConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/spinConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/spin)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/spinConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/spinConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/spinConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/spin)
if(SPIN_BUILD_EXPERIMENTS)
  foreach(name test bench screen direct scalar phases)
    if(name STREQUAL "scalar")
      add_executable(spin_feistel_scalar experiments/feistel/direct.cpp)
      target_compile_definitions(spin_feistel_scalar PRIVATE SPIN_FEISTEL_BATCH_ROUTE=0)
    else()
      add_executable(spin_feistel_${name} experiments/feistel/${name}.cpp)
    endif()
    target_link_libraries(spin_feistel_${name} PRIVATE spin::spin)
    target_compile_definitions(spin_feistel_${name} PRIVATE SPIN_BCH_AVX512=$<BOOL:${SPIN_ENABLE_AVX512}>)
    if(MSVC)
      target_compile_options(spin_feistel_${name} PRIVATE /arch:AVX2)
    else()
      target_compile_options(spin_feistel_${name} PRIVATE -mavx2 -mtune=${SPIN_TUNE})
    endif()
  endforeach()
  foreach(name test screen higher addresses flows optimize composed_test masks_test composed_screen affine_tail)
    add_executable(spin_bank_${name} experiments/permutation_bank/${name}.cpp)
    target_compile_features(spin_bank_${name} PRIVATE cxx_std_20)
    if(name STREQUAL "flows" OR name STREQUAL "optimize")
      target_link_libraries(spin_bank_${name} PRIVATE spin::spin)
      target_compile_definitions(spin_bank_${name} PRIVATE SPIN_BCH_AVX512=$<BOOL:${SPIN_ENABLE_AVX512}>)
    endif()
    if(name STREQUAL "optimize")
      target_compile_definitions(spin_bank_optimize PRIVATE SPIN_WORKSPACE_ROUTING_OPT=1)
    endif()
    if(name STREQUAL "addresses" OR name STREQUAL "flows" OR name STREQUAL "optimize")
      if(MSVC)
        target_compile_options(spin_bank_${name} PRIVATE /arch:AVX2)
      else()
        target_compile_options(spin_bank_${name} PRIVATE -mavx2 -mtune=${SPIN_TUNE})
      endif()
    endif()
  endforeach()
endif()
if(SPIN_BUILD_BENCHMARKS)
  add_executable(spin_paper_bench bench/paper.cpp)
  target_link_libraries(spin_paper_bench PRIVATE spin::spin)
  add_executable(spin_prepared_bench bench/prepared.cpp)
  target_link_libraries(spin_prepared_bench PRIVATE spin::spin)
  add_executable(spin_setup_bench bench/setup.cpp)
  target_compile_definitions(spin_setup_bench PRIVATE SPIN_BENCH_REUSE=1)
  target_link_libraries(spin_setup_bench PRIVATE spin::spin)
endif()
if(SPIN_BUILD_TESTS)
  enable_testing()
  find_package(Threads REQUIRED)
  add_executable(spin_api_test tests/api.cpp)
  target_link_libraries(spin_api_test PRIVATE spin::spin Threads::Threads)
  add_test(NAME spin_api COMMAND spin_api_test)
  add_executable(spin_known_answers tests/known_answers.cpp)
  target_link_libraries(spin_known_answers PRIVATE spin::spin)
  add_test(NAME spin_known_answers COMMAND spin_known_answers)
  add_executable(spin_setup_test tests/setup.cpp)
  target_compile_features(spin_setup_test PRIVATE cxx_std_20)
  add_test(NAME spin_setup COMMAND spin_setup_test)
  add_executable(spin_prepared_masks tests/prepared_masks.cpp)
  target_link_libraries(spin_prepared_masks PRIVATE spin::spin)
  target_compile_definitions(spin_prepared_masks PRIVATE SPIN_BCH_AVX512=$<BOOL:${SPIN_ENABLE_AVX512}>)
  add_test(NAME spin_prepared_masks COMMAND spin_prepared_masks)
  set_tests_properties(spin_prepared_masks PROPERTIES SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
  if(SPIN_BUILD_EXPERIMENTS)
    add_executable(spin_prepared_family tests/prepared_family.cpp)
    target_link_libraries(spin_prepared_family PRIVATE spin::spin)
    add_test(NAME spin_prepared_family COMMAND spin_prepared_family)
    add_test(NAME spin_feistel COMMAND spin_feistel_test)
    set_tests_properties(spin_feistel PROPERTIES SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
    add_test(NAME spin_bank COMMAND spin_bank_test)
    set_tests_properties(spin_bank PROPERTIES RUN_SERIAL TRUE)
    add_test(NAME spin_bank_flows COMMAND spin_bank_flows verify)
    set_tests_properties(spin_bank_flows PROPERTIES SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
    foreach(mode rolling split vector vector-split epoch composed composed-epoch composed-direct composed4 composed2 composed1 shift16 shift4 shift1 row-shift4 row-shift1 row-shift4-epoch row-shift1-epoch row-affine1 row-rotate1)
      add_test(NAME spin_bank_${mode} COMMAND spin_bank_optimize ${mode} verify)
      set_tests_properties(spin_bank_${mode} PROPERTIES SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
    endforeach()
    add_test(NAME spin_bank_composed_routing COMMAND spin_bank_composed_test)
    set_tests_properties(spin_bank_composed_routing PROPERTIES RUN_SERIAL TRUE)
    add_test(NAME spin_bank_masks COMMAND spin_bank_masks_test)
    set_tests_properties(spin_bank_masks PROPERTIES RUN_SERIAL TRUE)
  endif()
  set_tests_properties(spin_api spin_known_answers PROPERTIES SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
endif()
