\section{End-to-End Integrations}
\label{sec:integrations}

This section now focuses only on the dense validation pair.
The goal is not to route the argument through the most abstract framework interface,
but to write down the dense+dense first-moment bound directly and then plug in the
analytic outer and inner estimates proved in Sections~\ref{sec:outer-dense-logmem}
and~\ref{sec:inner-dense}.

\subsection{Direct dense+dense first-moment bound}
\label{sec:integration-dense-dense}

Take as outer code the dense log-memory construction of Section~\ref{sec:outer-dense-logmem},
with memory
\[
M=\lceil c_{\mathrm{out}}\log_2 n\rceil,
\]
and as inner code the dense time-varying recursive construction of Section~\ref{sec:inner-dense},
with memory
\[
m=\lceil \gamma \log_2 n\rceil.
\]
The resulting serial concatenation has one random interleaver and \(O(\log n)\) taps per symbol
in both the outer and inner encoders, hence \(O(n\log n)\) total encoding work.

\begin{theorem}[Direct dense+dense first-moment reduction]
\label{thm:dense-dense-direct-first-moment}
Fix \(\delta\in(0,1/2)\).
Let \(w_0=w_0(n)\in\{1,\dots,n\}\), and assume:
\begin{enumerate}
\item the outer code satisfies the cutoff event
\[
\Pr[d_{\min}(\Cone)<w_0]\le \varepsilon_{\mathrm{out}}(n);
\]
\item for every \(h\in\{w_0,\dots,n\}\),
\[
\EE[\enum_{\Cone,h}] \le A_h;
\]
\item for every \(h\in\{w_0,\dots,n\}\), if \(U\) is uniform on the Hamming slice
\(\{u\in\{0,1\}^n:\wt(u)=h\}\), then
\[
p_h(\delta)
:=
\Pr[\wt(\Enc_{\mathrm{in}}(U))\le \delta n\mid \wt(U)=h]
\le B_h.
\]
\end{enumerate}
Then the dense serial concatenation \(\Csc=\Ctwo\circ \Pi \circ \Cone\) satisfies
\begin{equation}
\label{eq:dense-dense-direct-first-moment}
\Pr[d_{\min}(\Csc)\le \delta n]
\;\le\;
\varepsilon_{\mathrm{out}}(n)
\;+\;
\sum_{h=w_0}^{n} A_h\,B_h.
\end{equation}
\end{theorem}

\begin{proof}
Let \(\mathcal{G}_{\mathrm{out}}\) denote the event \(d_{\min}(\Cone)\ge w_0\).
Then
\[
\Pr[d_{\min}(\Csc)\le \delta n]
\le
\Pr[\mathcal{G}_{\mathrm{out}}^c]
\;+\;
\Pr[d_{\min}(\Csc)\le \delta n\wedge \mathcal{G}_{\mathrm{out}}].
\]
The first term is \(\le \varepsilon_{\mathrm{out}}(n)\) by assumption.

Condition now on a realization of the outer code inside \(\mathcal{G}_{\mathrm{out}}\).
For each nonzero outer codeword \(c\in\Cone\) of weight \(h\), the random interleaver makes
\(\Pi(c)\) uniform on the weight-\(h\) slice, so
\[
\Pr[\wt(\Ctwo(\Pi(c)))\le \delta n\mid \wt(c)=h]
=
p_h(\delta).
\]
Applying Markov's inequality to the random variable counting low-weight nonzero concatenated codewords gives
\[
\Pr[d_{\min}(\Csc)\le \delta n\mid \Cone,\mathcal{G}_{\mathrm{out}}]
\le
\sum_{h=w_0}^{n}\enum_{\Cone,h}\,p_h(\delta).
\]
Taking expectation over the outer randomness and using the envelopes
\(\EE[\enum_{\Cone,h}]\le A_h\) and \(p_h(\delta)\le B_h\) yields
\eqref{eq:dense-dense-direct-first-moment}.
\end{proof}

\subsection{A complete coarse dense+dense theorem}
\label{sec:integration-dense-dense-coarse}

The dense sections already provide a complete analytic proof of linear distance if we are willing to use
their coarse geometric envelopes.
This gives a closed dense+dense theorem, even though it is not yet the sharp quantitative story.

\begin{theorem}[Dense outer + dense inner]
\label{thm:dense-dense-integration}
Fix \(\delta\in(0,1/2)\) and a geometric factor \(\rho\in(0,1)\) satisfying \(2\delta<\rho\).
Assume the dense outer supplies constants \(\alpha_{\mathrm{out}}>0\) and \(\beta_{\mathrm{out}}>1\)
such that, for all sufficiently large \(n\),
\[
\Pr[d_{\min}(\Cone)<\alpha_{\mathrm{out}}\log_2 n]=o(1)
\]
and
\[
\EE[\enum_{\Cone,h}] \le \beta_{\mathrm{out}}^{\,h}
\qquad
\bigl(h\ge \kappa_{\mathrm{cut}}\log_2 n\bigr),
\]
for some cutoff constant \(\kappa_{\mathrm{cut}}\in(0,\alpha_{\mathrm{out}}]\),
and assume the dense inner satisfies
\[
p_h(\delta)\le 3\rho^h
\qquad
\bigl(h\ge \kappa_{\mathrm{cut}}\log_2 n\bigr).
\]
If
\begin{equation}
\label{eq:dense-dense-coarse-compatibility}
\beta_{\mathrm{out}}\rho<1
\end{equation}
then
\[
\Pr[d_{\min}(\Csc)\le \delta n]=o(1).
\]
In particular, \(d_{\min}(\Csc)=\Omega(n)\) with high probability.
\end{theorem}

\begin{proof}
Set
\[
w_0:=\left\lceil \alpha_{\mathrm{out}}\log_2 n\right\rceil.
\]
By Theorem~\ref{thm:outer-dense-logdistance},
\[
\Pr[d_{\min}(\Cone)<w_0]=o(1).
\]
By Lemma~\ref{lem:outer-dense-spectrum},
\[
\EE[\enum_{\Cone,h}] \le \beta_{\mathrm{out}}^{\,h}
\qquad (h\ge w_0),
\]
and by Corollary~\ref{cor:dense-inner-constant-geometric},
\[
p_h(\delta)\le 3\rho^h
\qquad (h\ge w_0).
\]
Applying Theorem~\ref{thm:dense-dense-direct-first-moment},
\[
\Pr[d_{\min}(\Csc)\le \delta n]
\le
o(1)
\;+\;
3\sum_{h=w_0}^{n} (\beta_{\mathrm{out}}\rho)^h.
\]
Because \(\beta_{\mathrm{out}}\rho<1\), the geometric tail is
\[
\sum_{h=w_0}^{n} (\beta_{\mathrm{out}}\rho)^h
\le
O(1)\,(\beta_{\mathrm{out}}\rho)^{w_0}
=
n^{-\alpha_{\mathrm{out}}
\log_2(1/(\beta_{\mathrm{out}}\rho)) + o(1)}.
\]
Since \(\beta_{\mathrm{out}}\rho<1\), the exponent above is strictly negative, so the second term is \(o(1)\).
\end{proof}

\paragraph{What this theorem does and does not say.}
Theorem~\ref{thm:dense-dense-integration} is the cleanest fully analytic dense+dense result available from the current
lemmas.
It proves linear distance, but it is quantitatively crude because it compresses the outer and inner into global geometric
envelopes.
The remainder of this section keeps the same first-moment structure while exposing the sharper analytic profiles that the
dense outer and dense inner actually provide.

\begin{theorem}[Existence of a fully analytic dense+dense linear-distance regime]
\label{thm:dense-dense-existence}
Assume \(R_o=\tfrac12\), take \(M=\lceil c\log_2 n\rceil\) with \(c\ge 1\), and let \(\alpha_{\mathrm{out}}>0\) be the
constant from Theorem~\ref{thm:outer-dense-logdistance}.
Fix
\[
\kappa_{\mathrm{cut}}:=\alpha_{\mathrm{out}},
\]
and let \(\beta_{\mathrm{out}}=\beta_{\mathrm{out}}(c/\alpha_{\mathrm{out}})\) be the constant from
Lemma~\ref{lem:outer-dense-spectrum}.
Choose any \(\rho\in(0,1/\beta_{\mathrm{out}})\), then choose any
\[
\delta\in\Bigl(0,\frac{\rho}{2}\Bigr),
\]
and choose the dense-inner memory parameter \(\gamma\) so that
\[
\gamma>1+\alpha_{\mathrm{out}}\log_2\!\Bigl(\frac{1}{\rho}\Bigr).
\]
Then the resulting dense outer + dense inner concatenation satisfies
\[
\Pr[d_{\min}(\Csc)\le \delta n]=o(1).
\]
In particular, the dense+dense ensemble has linear minimum distance with high probability for some positive constant
\(\delta\).
\end{theorem}

\begin{proof}
Apply Theorem~\ref{thm:dense-dense-integration}.
Lemma~\ref{lem:outer-dense-spectrum} supplies the outer envelope above the cutoff
\(\kappa_{\mathrm{cut}}\log_2 n=\alpha_{\mathrm{out}}\log_2 n\).
Because \(\delta<\rho/2\) and \(\gamma>1+\alpha_{\mathrm{out}}\log_2(1/\rho)\),
Corollary~\ref{cor:dense-inner-constant-geometric} supplies
\[
p_h(\delta)\le 3\rho^h
\qquad
(h\ge \alpha_{\mathrm{out}}\log_2 n).
\]
Finally, \(\rho<1/\beta_{\mathrm{out}}\) is exactly the compatibility condition
\eqref{eq:dense-dense-coarse-compatibility}.
\end{proof}

\subsection{The sharp dense+dense objective}
\label{sec:integration-dense-dense-piecewise}

For the dense outer, Corollary~\ref{cor:outer-dense-spectrum-piecewise} gives the exact expected spectrum
\[
A_h^{\mathrm{dense}}
:=
\sum_{\ell=1}^{k} N_\ell\,
\binom{r(\ell+M)}{h}\,2^{-r(\ell+M)}
\qquad (1\le h\le n),
\]
with \(N_1=k\) and \(N_\ell=(k-\ell+1)2^{\ell-2}\) for \(\ell\ge 2\).

For the dense inner, Corollary~\ref{cor:dense-inner-piecewise} gives the envelope
\begin{equation}
\label{eq:dense-dense-inner-profile}
B_h^{\mathrm{dense}}(\delta;\varepsilon,\xi,m)
\;:=\;
\min\!\left\{
(1-\varepsilon)^h + n2^{-m} + q_{\delta,\varepsilon}(n),\ 
\min_{1\le r\le h}
\Bigl(
\mathrm{RunTail}_n(h,r) + (L2^{-m})^r + r\,b_{\delta,\xi}(n)
\Bigr)
\right\},
\end{equation}
where
\[
L=\lceil(2+\xi)\delta n\rceil,
\quad
q_{\delta,\varepsilon}(n)
:=
\PP\!\left[\ \Binomial(\lfloor(1-\varepsilon)n\rfloor,\tfrac12)\le \delta n\ \right],
\]
\[
b_{\delta,\xi}(n)
:=
\PP\!\left[\ \Binomial(L,\tfrac12)\le \delta n\ \right].
\]

\begin{theorem}[Explicit dense+dense first-moment objective]
\label{thm:dense-dense-explicit}
Fix \(\delta\in(0,1/2)\), \(\varepsilon\in(0,1)\), and \(\xi>0\).
Let
\[
w_0:=\left\lceil \alpha_{\mathrm{out}}\log_2 n\right\rceil,
\]
where \(\alpha_{\mathrm{out}}\) is the logarithmic-distance constant from
Theorem~\ref{thm:outer-dense-logdistance}.
Then
\begin{equation}
\label{eq:dense-dense-explicit-objective}
\Pr[d_{\min}(\Csc)\le \delta n]
\;\le\;
\Pr[d_{\min}(\Cone)<w_0]
\;+\;
\sum_{h=w_0}^{n}
A_h^{\mathrm{dense}}\,
B_h^{\mathrm{dense}}(\delta;\varepsilon,\xi,m).
\end{equation}
\end{theorem}

\begin{proof}
Apply Theorem~\ref{thm:dense-dense-direct-first-moment} with
\[
A_h=A_h^{\mathrm{dense}}
\]
from Corollary~\ref{cor:outer-dense-spectrum-piecewise} and
\[
B_h=B_h^{\mathrm{dense}}(\delta;\varepsilon,\xi,m)
\]
from Corollary~\ref{cor:dense-inner-piecewise}.
\end{proof}

\paragraph{This is the dense+dense optimization target.}
Equation~\eqref{eq:dense-dense-explicit-objective} is the quantity that the dense lemmas now support directly.
It is still a pure first-moment argument, but it no longer hides the actual outer and inner analytic profiles behind
one global envelope.
This is the right bridge between the proof and the enumerator-guided finite-length experiments.

\subsection{Linear-weight exponent criterion}
\label{sec:integration-dense-dense-linear}

The explicit objective above is still a finite-\(n\) sum.
At rate \(1/2\), the linear-weight terms can be controlled analytically by pairing the dense-outer exponent
\(\Phi_{1/2}\) with the dense-inner linear-weight exponent from
Proposition~\ref{prop:dense-inner-linear-exponent}.

\begin{theorem}[Linear-weight dense+dense criterion at rate \(\tfrac12\)]
\label{prop:dense-dense-linear-window}
Assume the outer rate is \(R_o=\tfrac12\), so \(r=2\).
Fix constants
\[
\delta\in(0,1/2),
\qquad
\omega_0\in(0,1/2),
\qquad
\xi>0,
\qquad
\theta\in(0,1-\omega_0),
\]
and let
\[
M=\Theta(\log n),
\qquad
m=\gamma\log_2 n
\]
with \(\gamma>1\).
Define
\[
E_{\mathrm{bin}}(\delta,\xi)
\;:=\;
(2+\xi)\delta\left(1-H_2\!\Bigl(\frac{1}{2+\xi}\Bigr)\right)
\]
and
\[
E_{\mathrm{in}}(\omega;\theta,\delta,\xi)
\;:=\;
\min\!\left\{
\Psi_{\mathrm{run}}^{(w)}(\omega,\theta),\
E_{\mathrm{bin}}(\delta,\xi)
\right\}.
\]
If
\begin{equation}
\label{eq:dense-dense-linear-gap}
\eta_{\mathrm{gap}}
\;:=\;
\inf_{\omega\in[\omega_0,\;1/2]}
\left(
E_{\mathrm{in}}(\omega;\theta,\delta,\xi)-\Phi_{1/2}(\omega)
\right)
>0,
\end{equation}
then
\begin{equation}
\label{eq:dense-dense-linear-sum}
\sum_{h=\lceil \omega_0 n\rceil}^{\lfloor n/2\rfloor}
A_h^{\mathrm{dense}}\,
B_h^{\mathrm{dense}}(\delta;\varepsilon,\xi,m)
\;=\;
o(1).
\end{equation}
\end{theorem}

\begin{proof}
Fix \(h=\lfloor \omega n\rfloor\) with \(\omega\in[\omega_0,1/2]\).
By Proposition~\ref{prop:outer-dense-rate-half-exponent},
\[
A_h^{\mathrm{dense}}
\;\le\;
n^{O(1)}\,2^{n\Phi_{1/2}(\omega)}.
\]
By Proposition~\ref{prop:dense-inner-linear-exponent}, with the choice \(r=\lfloor \theta h\rfloor\),
\[
B_h^{\mathrm{dense}}(\delta;\varepsilon,\xi,m)
\;\le\;
n^{O(1)}\,2^{-nE_{\mathrm{in}}(\omega;\theta,\delta,\xi)}.
\]
Hence
\[
A_h^{\mathrm{dense}}\,B_h^{\mathrm{dense}}(\delta;\varepsilon,\xi,m)
\;\le\;
n^{O(1)}\,2^{-n\left(E_{\mathrm{in}}(\omega;\theta,\delta,\xi)-\Phi_{1/2}(\omega)\right)}.
\]
Under \eqref{eq:dense-dense-linear-gap}, every summand is at most
\[
n^{O(1)}\,2^{-n\eta_{\mathrm{gap}}}.
\]
There are only \(O(n)\) values of \(h\) in the range \([\omega_0 n,n/2]\), so the sum in
\eqref{eq:dense-dense-linear-sum} is \(o(1)\).
\end{proof}

\paragraph{What remains after the linear-weight window.}
The linear-window criterion isolates the dense+dense question that matters for quantitatively meaningful distances:
can the dense-inner exponent beat the dense-outer exponent on the entire weight interval relevant to the target \(\delta\)?
The low and intermediate weights remain controlled by the coarse theorem
\ref{thm:dense-dense-integration}, while Proposition~\ref{prop:dense-dense-linear-window}
is the sharper tool for the GV-relevant range.
