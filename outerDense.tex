\section{Random sliding dense outer (log-memory)}
\label{sec:outer-dense-logmem}

For the dense validation pair, the outer code is now taken to be a \emph{systematic}
banded convolutional map.
This is the analytically natural object suggested by the enumerator experiments:
the systematic stream prevents the dangerous collapse of a moderate message weight into a tiny
intermediate weight, while the parity stream retains the same local dense-tap structure.

\subsection{Construction}
\label{sec:outer-dense-construction}

We work at outer rate \(R_o=\tfrac12\).
Let \(k=n/2\) and let \(M=\lceil c\log_2 n\rceil\) for a constant \(c\ge 1\).

Given information bits \(x_1,\dots,x_k\in\F_2\), append zero-termination
\[
x_{k+1}=\cdots=x_{k+M}=0
\qquad\text{and}\qquad
x_t=0\ \text{for }t\le 0.
\]
For each time \(t\in[k]\), sample an independent random vector
\[
a_t\gets \F_2^M,
\qquad
g_t:=(1,a_t)\in \F_2^{M+1},
\]
and define the parity output
\begin{equation}
\label{eq:outer-dense-parity}
p_t \;:=\; \big\langle g_t,\,(x_t,x_{t-1},\dots,x_{t-M})\big\rangle.
\end{equation}
The length-\(n\) outer codeword is the systematic-parity concatenation
\[
\Enc_{\Cone}(x)
:=
(x_1,p_1,\dots,x_k,p_k)\in\F_2^{2k}.
\]
Encoding cost is \(O(kM)=O(n\log n)\).

The fixed leading parity tap is the outer analogue of the fixed-tap dense recursive inner. In particular, a weight-one
message can no longer have zero parity response: the diagonal parity coordinate is forced to be one.
The detailed tiny-weight parity law for this fixed tap is preserved in the random sliding outer exploration note; the
main dense+dense theorem below only needs the span generating bound and the resulting spectrum envelopes.

\paragraph{Spectrum notation.}
For \(h\in\{0,\dots,n\}\), write
\[
\enum_{\Cone,h}
:=
\bigl|\{c\in\Cone:\wt(c)=h\}\bigr|.
\]
As in the framework, \(\EE[\enum_{\Cone,h}]\) denotes expectation over the random taps.

\subsection{A fixed-word parity law}
\label{sec:outer-dense-fixedword}

\begin{definition}[Span]
For a nonzero input \(x\in\F_2^k\), let \(a\) be the first index with \(x_a=1\) and \(b\) the last index with \(x_b=1\).
The span of \(x\) is \(\ell:=b-a+1\).
\end{definition}

\begin{lemma}[Parity generating bound for a fixed span]
\label{lem:outer-span-binomial}
Fix a nonzero input \(x\in\F_2^k\) of span \(\ell\) and weight \(w\).
Then exactly \(\ell+M\) parity positions have a nonzero window. For every \(z\in(0,1)\),
\[
\EE\!\left[z^{\wt(\Enc_{\Cone}(x))}\right]
\le
z^w\left(\frac{1+z}{2}\right)^{\ell+M}.
\]
\end{lemma}

\begin{proof}
If the window \((x_t,\dots,x_{t-M})\) is nonzero, then
\(\langle g_t,(x_t,\dots,x_{t-M})\rangle\) is either a nontrivial linear functional of the random coordinates
\(a_t\), hence uniform in \(\F_2\), or else the window is supported only on the fixed leading coordinate, in which case
the parity bit is deterministically one.
Independence across \(t\) follows from independence of the tap vectors \(g_t\).
For a fair parity bit, the contribution to the \(z\)-generating function is \((1+z)/2\). For a deterministic one, it
is \(z\), which is at most \((1+z)/2\) when \(z\in(0,1)\). Multiplying over the \(\ell+M\) active parity positions and
including the systematic factor \(z^w\) proves the bound.
\end{proof}

\subsection{A global geometric spectrum envelope}
\label{sec:outer-dense-spectrum}

The systematic structure makes it possible to control the \emph{entire} expected outer spectrum by a single geometric
bound.
The proof is a weighted generating-function argument over message spans.

\begin{theorem}[Systematic dense-outer generating-function bound]
\label{thm:outer-dense-generating}
Let
\[
A_h := \EE[\enum_{\Cone,h}],
\qquad
W_{\mathrm{out}}(z):=\sum_{h\ge 1} A_h z^h.
\]
Fix any \(z\in(0,\sqrt2-1)\), and define
\[
a(z):=\frac{1+z}{2},
\qquad
q(z):=\frac{(1+z)^2}{2}.
\]
Then \(q(z)<1\), and there exists a constant \(C_{\mathrm{out}}(z)>0\) such that
\begin{equation}
\label{eq:outer-dense-gf-bound}
W_{\mathrm{out}}(z)
\;\le\;
C_{\mathrm{out}}(z)\, k\, a(z)^M.
\end{equation}
\end{theorem}

\begin{proof}
For a fixed nonzero input \(x\), Lemma~\ref{lem:outer-span-binomial} gives
\[
\EE\!\left[z^{\wt(\Enc_{\Cone}(x))}\right]
=
z^{\wt(x)}\left(\frac{1+z}{2}\right)^{\operatorname{span}(x)+M}.
\]
Summing over all nonzero inputs,
\[
W_{\mathrm{out}}(z)
=
\sum_{x\neq 0}
z^{\wt(x)}a(z)^{\operatorname{span}(x)+M}.
\]

Group the inputs by span.
For span \(1\), there are exactly \(k\) inputs, each of weight \(1\), so the total contribution is
\[
k\,z\,a(z)^{M+1}.
\]

For span \(\ell\ge 2\), choose the start position in \(k-\ell+1\) ways.
The two endpoints of the span are forced to be \(1\), while the \(\ell-2\) interior bits are arbitrary.
Hence the sum of \(z^{\wt(x)}\) over all inputs of span \(\ell\) is
\[
(k-\ell+1)\,z^2(1+z)^{\ell-2}.
\]
Therefore
\[
W_{\mathrm{out}}(z)
\le
kza(z)^{M+1}
\sum_{\ell=2}^{k}
(k-\ell+1)\,z^2(1+z)^{\ell-2}a(z)^{\ell+M}.
\]
Using
\[
(1+z)^{\ell-2}a(z)^\ell
=(1+z)^{-2}\left(\frac{(1+z)^2}{2}\right)^\ell
=(1+z)^{-2}q(z)^\ell,
\]
we get
\[
W_{\mathrm{out}}(z)
\le
kza(z)^{M+1}
z^2a(z)^M(1+z)^{-2}
\sum_{\ell=2}^{k}(k-\ell+1)q(z)^\ell.
\]
Since \(k-\ell+1\le k\) and \(q(z)<1\),
\[
\sum_{\ell=2}^{k}(k-\ell+1)q(z)^\ell
\le
k\sum_{\ell=2}^{\infty}q(z)^\ell.
\]
The geometric series converges, so both terms are \(O_z(k a(z)^M)\), proving \eqref{eq:outer-dense-gf-bound}.
\end{proof}

\begin{corollary}[Global geometric spectrum envelope]
\label{lem:outer-dense-spectrum}
Fix any \(z\in(0,\sqrt2-1)\) and choose
\[
c>\frac{1}{\log_2(1/a(z))}
=
\frac{1}{\log_2(2/(1+z))}.
\]
Then for all sufficiently large \(n\),
\[
\EE[\enum_{\Cone,h}] \le z^{-h}
\qquad\text{for all }h\in\{1,\dots,n\}
\]
In particular, one may take \(\beta_{\mathrm{out}}=z^{-1}\) in the dense+dense integration theorem.
\end{corollary}

\begin{proof}
By Theorem~\ref{thm:outer-dense-generating},
\[
W_{\mathrm{out}}(z)\le C_{\mathrm{out}}(z)\,k\,a(z)^M.
\]
Since \(k=n/2\) and \(M=\lceil c\log_2 n\rceil\),
\[
k\,a(z)^M
\le
n^{1-c\log_2(1/a(z))}.
\]
The assumption on \(c\) makes the exponent negative, so in fact \(W_{\mathrm{out}}(z)\to 0\) as \(n\to\infty\).
Hence \(W_{\mathrm{out}}(z)\le 1\) for all sufficiently large \(n\).
Now coefficient extraction gives
\[
A_h \le \frac{W_{\mathrm{out}}(z)}{z^h}\le z^{-h}.
\]
\end{proof}

\subsection{A linear-weight outer exponent}
\label{sec:outer-dense-linear-exponent}

The global geometric envelope is enough for the baseline theorem, but for sharper dense+dense claims we also want
the linear-weight exponent suggested by the exact span decomposition.

\begin{theorem}[Systematic dense outer: linear-weight spectrum exponent]
\label{thm:outer-dense-linear-exponent}
Let \(h=\lfloor \eta n\rfloor\) with \(\eta\in(0,1)\), and write
\[
A_h:=\EE[\enum_{\Cone,h}].
\]
Define
\begin{equation}
\label{eq:outer-dense-linear-rate-function}
\Phi_{\mathrm{out}}^{\mathrm{sys}}(\eta)
\;:=\;
\sup_{\eta/2\le \lambda\le 1/2}
\lambda
\left[
2H_2\!\Bigl(\frac{\eta}{2\lambda}\Bigr)
-1
\right].
\end{equation}
Then
\begin{equation}
\label{eq:outer-dense-linear-exponent}
A_h
\;\le\;
n^{O(1)}\,2^{\,n\,\Phi_{\mathrm{out}}^{\mathrm{sys}}(\eta)}.
\end{equation}
\end{theorem}

\begin{proof}
For \(w\ge 2\) and span \(\ell\ge w\), the number of message words of weight \(w\) and span \(\ell\) is
\[
(k-\ell+1)\binom{\ell-2}{w-2}.
\]
By Lemma~\ref{lem:outer-span-binomial}, each such word contributes
\[
2^{-(\ell+M)}\binom{\ell+M}{h-w}
\]
in expectation to \(A_h\).
Thus
\[
A_h
\le
\sum_{w=2}^{k}\sum_{\ell=w}^{k}
(k-\ell+1)\binom{\ell-2}{w-2}\,
2^{-(\ell+M)}\binom{\ell+M}{h-w}
\;+\;k\,2^{-M}\binom{M+1}{h-1},
\]
where the final term is the special case \(w=\ell=1\).

The \(w=1\) term is at most \(n^{O(1)}2^{o(n)}\), so it is negligible on the exponential scale.
For the double sum, group first by span \(\ell\).
For each fixed \(\ell\ge 2\), Vandermonde's identity gives
\[
\sum_{w=2}^{\min\{h,\ell\}}
\binom{\ell-2}{w-2}\binom{\ell+M}{h-w}
=
\binom{2\ell+M-2}{h-2}.
\]
Hence
\[
A_h
\le
\sum_{\ell=2}^{k}
(k-\ell+1)\,2^{-(\ell+M)}\binom{2\ell+M-2}{h-2}
\;+\;n^{O(1)}2^{o(n)}.
\]

Now write
\[
\ell=\lfloor \lambda n\rfloor,
\qquad
h=\lfloor \eta n\rfloor.
\]
If \(h>2\ell+M-2\), the binomial coefficient vanishes, so asymptotically we must have
\[
\eta/2\le \lambda\le 1/2.
\]
Using Stirling bounds and \(M=O(\log n)=o(n)\),
\[
\binom{2\ell+M-2}{h-2}
\le
n^{O(1)}\,2^{(2\ell+M-2)H_2((h-2)/(2\ell+M-2))}
\le
n^{O(1)}\,2^{2\lambda n\,H_2(\eta/(2\lambda))+o(n)}.
\]
Multiplying by \(2^{-(\ell+M)}\) gives
\[
n^{O(1)}\,
2^{\,n\lambda\left[
2H_2(\eta/(2\lambda))-1
\right]+o(n)}.
\]
Summing over the at most \(n\) admissible spans \(\ell\) only changes the polynomial prefactor, yielding
\eqref{eq:outer-dense-linear-exponent}.
\end{proof}

\begin{corollary}[Explicit formula for the outer exponent]
\label{cor:outer-dense-highweight-exponent}
Let
\[
\eta_{\mathrm{crit}}:=1-\frac{1}{\sqrt2}.
\]
Then for every \(\eta\in(0,1)\),
\[
\Phi_{\mathrm{out}}^{\mathrm{sys}}(\eta)
=
\begin{cases}
\eta \log_2(1+\sqrt2), & 0<\eta\le \eta_{\mathrm{crit}},\\[1ex]
H_2(\eta)-\frac12, & \eta_{\mathrm{crit}}\le \eta<1.
\end{cases}
\]
\end{corollary}

\begin{proof}
Fix \(\eta\in(0,1)\).
Set
\[
t:=\frac{\eta}{2\lambda},
\qquad
g(t):=\frac{H_2(t)-1/2}{t}.
\]
Then \(t\in[\eta,1]\) and
\[
\lambda\left[
2H_2\!\Bigl(\frac{\eta}{2\lambda}\Bigr)-1
\right]
=
\eta\,g(t).
\]
A direct derivative computation gives
\[
g'(t)
=
\frac{\log_2(1-t)+1/2}{t^2}.
\]
Thus \(g\) is increasing on \((0,\eta_{\mathrm{crit}}]\) and decreasing on \([\eta_{\mathrm{crit}},1)\).

If \(\eta\le \eta_{\mathrm{crit}}\), then the feasible interval \([\eta,1]\) contains \(\eta_{\mathrm{crit}}\), so the supremum
is attained at \(t=\eta_{\mathrm{crit}}\). Hence
\[
\Phi_{\mathrm{out}}^{\mathrm{sys}}(\eta)
=
\eta\,g(\eta_{\mathrm{crit}}).
\]
Since \(\log_2(1-\eta_{\mathrm{crit}})=-1/2\),
\[
H_2(\eta_{\mathrm{crit}})-\frac12
=
-\eta_{\mathrm{crit}}\log_2(\eta_{\mathrm{crit}})
-(1-\eta_{\mathrm{crit}})\log_2(1-\eta_{\mathrm{crit}})
-\frac12
=
\eta_{\mathrm{crit}}\!\left(-\log_2(\eta_{\mathrm{crit}})-\frac12\right).
\]
Therefore
\[
g(\eta_{\mathrm{crit}})
=
-\log_2(\eta_{\mathrm{crit}})-\frac12
=
-\log_2\!\Bigl(\frac{\sqrt2-1}{\sqrt2}\Bigr)-\frac12
=
-\log_2(\sqrt2-1)
=
\log_2(1+\sqrt2),
\]
which proves the first branch.

If \(\eta\ge \eta_{\mathrm{crit}}\), then every admissible \(t\) lies in \([\eta,1]\subseteq [\eta_{\mathrm{crit}},1]\),
so \(g\) is decreasing throughout the feasible interval.
Therefore the supremum is attained at the smallest admissible \(t\), namely \(t=\eta\), which corresponds to
\(\lambda=1/2\).
Substituting \(\lambda=1/2\) gives
\[
\Phi_{\mathrm{out}}^{\mathrm{sys}}(\eta)
=
\frac12\left[2H_2(\eta)-1\right]
=
H_2(\eta)-\frac12.
\qedhere
\]
\end{proof}

\begin{corollary}[Low-linear-weight outer envelope]
\label{cor:outer-dense-lowlinear-envelope}
For every \(h\le \eta_{\mathrm{crit}}n\),
\[
\EE[\enum_{\Cone,h}]
\le
n^{O(1)}(1+\sqrt2)^h.
\]
\end{corollary}

\begin{proof}
Write \(h=\lfloor \eta n\rfloor\) with \(\eta\le \eta_{\mathrm{crit}}\).
Theorem~\ref{thm:outer-dense-linear-exponent} and Corollary~\ref{cor:outer-dense-highweight-exponent} give
\[
\EE[\enum_{\Cone,h}]
\le
n^{O(1)}2^{n\eta\log_2(1+\sqrt2)}
\le
n^{O(1)}(1+\sqrt2)^h,
\]
after absorbing the floor/ceiling discrepancy into the polynomial factor.
\end{proof}
