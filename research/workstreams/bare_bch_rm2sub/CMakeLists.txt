cmake_minimum_required(VERSION 3.20)
project(BareBchRm2Sub LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SPIN_VENDOR "${REPO_ROOT}/constructions/riffle_parityfanout31x33_bchperm_transpose_bitshuffle_splitstate_preaddmul_rm2sub_t128_s19/implementation/vendor" CACHE PATH "Standalone cryptoTools headers")
set(SPIN_ARCH "znver4" CACHE STRING "GCC architecture")
option(SPIN_GROUPED_A "Use measured optimized nibble grouping for S19" ON)
option(SPIN_WORKSPACE_ROUTING_OPT "Opt in to Linux large-page workspace hints and quarter-rate write-prefetch for K>=2^18" OFF)
add_library(bare_spin STATIC Spin.cpp generated/BchCircuit.cpp
    ../rate_quarter_bch/implementation/generated/QuarterCircuit.cpp)
target_compile_definitions(bare_spin PRIVATE SPIN_GROUPED_A=$<BOOL:${SPIN_GROUPED_A}>)
target_compile_definitions(bare_spin PRIVATE SPIN_WORKSPACE_ROUTING_OPT=$<BOOL:${SPIN_WORKSPACE_ROUTING_OPT}>)
target_include_directories(bare_spin PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${SPIN_VENDOR}")
add_executable(spin_correctness correctness.cpp)
target_link_libraries(spin_correctness PRIVATE bare_spin)
add_executable(spin_benchmark benchmark.cpp)
target_link_libraries(spin_benchmark PRIVATE bare_spin)
add_executable(quarter_correctness ../rate_quarter_bch/implementation/correctness.cpp)
target_link_libraries(quarter_correctness PRIVATE bare_spin)
foreach(t bare_spin spin_correctness spin_benchmark quarter_correctness)
    if(MSVC)
        target_compile_options(${t} PRIVATE /arch:AVX2)
    else()
        target_compile_options(${t} PRIVATE "-march=${SPIN_ARCH}" -mavx2 -mpclmul -mvpclmulqdq)
    endif()
endforeach()
enable_testing()
add_test(NAME spin_correctness COMMAND spin_correctness)
add_test(NAME quarter_correctness COMMAND quarter_correctness)
