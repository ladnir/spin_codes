cmake_minimum_required(VERSION 3.20)
project(spin_rs16x8_border_experiment LANGUAGES CXX)
get_filename_component(SPIN_REPOSITORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(SPIN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPIN_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(SPIN_BUILD_EXPERIMENTS OFF CACHE BOOL "" FORCE)
set(SPIN_ENABLE_AVX512 ON CACHE BOOL "" FORCE)
add_subdirectory("${SPIN_REPOSITORY}/spin" spin)
add_executable(spin_rs16x8_border driver.cpp RsBorderSetup.cpp RsBorderScalar.cpp RsBorderFast.cpp RsBorderFused.cpp RsBorderTernary.cpp RsBorderEarly.cpp
  ../rs16x8/RsWideSetup.cpp ../rs16x8/RsWideScalar.cpp ../rs16x8/RsWideFast.cpp ../rs16x8/RsWideOuterVariants.cpp)
target_compile_features(spin_rs16x8_border PRIVATE cxx_std_20)
target_link_libraries(spin_rs16x8_border PRIVATE spin::spin)
if(MSVC)
  set_source_files_properties(RsBorderFast.cpp RsBorderFused.cpp RsBorderTernary.cpp RsBorderEarly.cpp ../rs16x8/RsWideFast.cpp ../rs16x8/RsWideOuterVariants.cpp PROPERTIES COMPILE_OPTIONS /arch:AVX512)
  set_source_files_properties(RsBorderFast.cpp RsBorderFused.cpp RsBorderTernary.cpp RsBorderEarly.cpp ../rs16x8/RsWideFast.cpp ../rs16x8/RsWideOuterVariants.cpp PROPERTIES COMPILE_DEFINITIONS SPIN_PACKET_ENABLE_VBMI=1)
else()
  target_compile_options(spin_rs16x8_border PRIVATE -fno-strict-aliasing -fno-lto -mtune=${SPIN_TUNE})
  set_source_files_properties(RsBorderFast.cpp RsBorderFused.cpp RsBorderTernary.cpp RsBorderEarly.cpp ../rs16x8/RsWideFast.cpp ../rs16x8/RsWideOuterVariants.cpp PROPERTIES COMPILE_OPTIONS
    "-mavx2;-mavx512f;-mavx512dq;-mavx512vl;-mavx512bw;-mavx512vbmi;-mgfni")
endif()
enable_testing()
add_test(NAME rs16x8_border_correctness COMMAND spin_rs16x8_border check)
set_tests_properties(rs16x8_border_correctness PROPERTIES RUN_SERIAL TRUE SKIP_RETURN_CODE 77)
