cmake_minimum_required(VERSION 3.20)
project(Weight5HalfRate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
set(BARE "${ROOT}/workstreams/bare_bch_rm2sub")
set(QUARTER "${ROOT}/workstreams/rate_quarter_bch/implementation/generated")
set(VENDOR "${ROOT}/constructions/riffle_parityfanout31x33_bchperm_transpose_bitshuffle_splitstate_preaddmul_rm2sub_t128_s19/implementation/vendor")
function(configure target)
    target_include_directories(${target} PRIVATE "${BARE}" "${QUARTER}" "${VENDOR}")
    target_compile_options(${target} PRIVATE -march=znver4 -mavx2 -mpclmul -mvpclmulqdq)
endfunction()
add_library(outers STATIC "${BARE}/generated/BchCircuit.cpp" "${QUARTER}/QuarterCircuit.cpp")
configure(outers)
enable_testing()
foreach(name baseline baseline_tuned sparse sparse_pages sparse_tuned masked_tuned shared_tuned grouped_tuned sparse_pf128)
    set(pages 0)
    set(masked 0)
    set(shared 0)
    set(write 0)
    set(prefetch 32)
    if(name STREQUAL baseline)
        set(source "${BARE}/Spin.cpp")
    elseif(name STREQUAL baseline_tuned)
        set(source BaselineTuned.cpp)
    else()
        set(source Weight5Spin.cpp)
    endif()
    if(name MATCHES "(tuned|pages|pf128)$")
        set(pages 1)
    endif()
    if(name MATCHES "(tuned|pf128)$")
        set(write 1)
    endif()
    if(name MATCHES "^masked")
        set(masked 1)
    endif()
    if(name MATCHES "^shared")
        set(shared 1)
    endif()
    if(name MATCHES "^grouped")
        set(shared 2)
    endif()
    if(name MATCHES "pf128$")
        set(prefetch 128)
    endif()
    add_library(${name} STATIC "${source}")
    configure(${name})
    target_compile_definitions(${name} PRIVATE SPIN_GROUPED_A=1 SPIN_WORKSPACE_ROUTING_OPT=${pages}
        W5_MASKED=${masked} W5_SHARED_EMISSION=${shared} W5_WRITE_PREFETCH=${write} W5_PREFETCH=${prefetch})
    target_link_libraries(${name} PUBLIC outers)
    add_executable(${name}_bench "${BARE}/benchmark.cpp")
    configure(${name}_bench)
    target_link_libraries(${name}_bench PRIVATE ${name})
    add_executable(${name}_test correctness.cpp)
    configure(${name}_test)
    target_link_libraries(${name}_test PRIVATE ${name})
    add_test(NAME ${name}_test COMMAND ${name}_test)
endforeach()
foreach(shared 0 1 2)
    add_executable(inner_identity_${shared}_test inner_identity.cpp)
    configure(inner_identity_${shared}_test)
    target_compile_definitions(inner_identity_${shared}_test PRIVATE W5_SHARED_EMISSION=${shared})
    add_test(NAME inner_identity_${shared}_test COMMAND inner_identity_${shared}_test)
endforeach()
