cmake_minimum_required(VERSION 3.20)
project(SpinScheduling LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(BARE "${ROOT}/workstreams/bare_bch_rm2sub")
set(BASE "${ROOT}/workstreams/inner_design/asymmetric/bch256/weight5/implementation")
set(QUARTER "${ROOT}/workstreams/rate_quarter_bch/implementation/generated")
set(VENDOR "${ROOT}/constructions/riffle_parityfanout31x33_bchperm_transpose_bitshuffle_splitstate_preaddmul_rm2sub_t128_s19/implementation/vendor")
function(configure target)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${BASE}" "${BARE}" "${QUARTER}" "${VENDOR}")
  target_compile_options(${target} PRIVATE -march=znver4 -mavx2 -mpclmul -mvpclmulqdq)
endfunction()
add_library(outers STATIC "${BARE}/generated/BchCircuit.cpp" "${QUARTER}/QuarterCircuit.cpp")
configure(outers)
enable_testing()
function(candidate name group pf decode lead)
  set(source generated/ScheduledSpin.cpp)
  if(name STREQUAL control)
    set(source "${BASE}/Weight5Spin.cpp")
  endif()
  add_library(${name} STATIC "${source}")
  configure(${name})
  target_compile_definitions(${name} PRIVATE SPIN_GROUPED_A=1 SPIN_WORKSPACE_ROUTING_OPT=1
    W5_MASKED=0 W5_SHARED_EMISSION=0 W5_WRITE_PREFETCH=0 W5_PREFETCH=${pf}
    SCHEDULE_GROUP=${group} SCHEDULE_DECODE=${decode} SCHEDULE_EMIT_LEAD=${lead})
  if(name STREQUAL profile)
    target_compile_definitions(${name} PRIVATE W5_PROFILE=1)
  endif()
  target_link_libraries(${name} PUBLIC outers)
  add_executable(${name}_bench benchmark.cpp)
  configure(${name}_bench)
  target_link_libraries(${name}_bench PRIVATE ${name})
  add_executable(${name}_test "${BASE}/correctness.cpp")
  configure(${name}_test)
  target_link_libraries(${name}_test PRIVATE ${name})
  add_test(NAME ${name} COMMAND ${name}_test)
endfunction()
