cmake_minimum_required(VERSION 3.20)
project(SpinBchScheduling LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(BARE "${ROOT}/workstreams/bare_bch_rm2sub")
set(BASE "${ROOT}/workstreams/inner_design/asymmetric/bch256/weight5/implementation")
set(QUARTER "${ROOT}/workstreams/rate_quarter_bch/implementation/generated")
set(VENDOR "${ROOT}/constructions/riffle_parityfanout31x33_bchperm_transpose_bitshuffle_splitstate_preaddmul_rm2sub_t128_s19/implementation/vendor")
function(configure target)
  target_include_directories(${target} PRIVATE "${BARE}" "${BASE}" "${QUARTER}" "${VENDOR}")
  target_compile_options(${target} PRIVATE -march=znver4 -mavx2 -mpclmul -mvpclmulqdq)
endfunction()
add_library(spin STATIC "${BASE}/Weight5Spin.cpp" "${QUARTER}/QuarterCircuit.cpp")
configure(spin)
target_compile_definitions(spin PRIVATE SPIN_GROUPED_A=1 SPIN_WORKSPACE_ROUTING_OPT=1 W5_MASKED=0 W5_SHARED_EMISSION=0 W5_WRITE_PREFETCH=0 W5_PREFETCH=32)
enable_testing()
add_library(spin_four STATIC generated/SpinFour.cpp "${QUARTER}/QuarterCircuit.cpp")
configure(spin_four)
target_compile_definitions(spin_four PRIVATE SPIN_GROUPED_A=1 SPIN_WORKSPACE_ROUTING_OPT=1 W5_MASKED=0 W5_SHARED_EMISSION=0 W5_WRITE_PREFETCH=0 W5_PREFETCH=32)
add_library(spin_packedfour STATIC generated/SpinPackedFour.cpp "${QUARTER}/QuarterCircuit.cpp")
configure(spin_packedfour)
target_compile_definitions(spin_packedfour PRIVATE SPIN_GROUPED_A=1 SPIN_WORKSPACE_ROUTING_OPT=1 W5_MASKED=0 W5_SHARED_EMISSION=0 W5_WRITE_PREFETCH=0 W5_PREFETCH=32)
foreach(name control dfs chunk32 synth32 synth16 single share3 share4 share8 four packedfour packedshare3 packedshare4)
  set(spin_target spin)
  if(name STREQUAL four)
    set(spin_target spin_four)
  elseif(name MATCHES "^packed")
    set(spin_target spin_packedfour)
  endif()
  set(source generated/${name}.cpp)
  if(name STREQUAL control)
    set(source "${BARE}/generated/BchCircuit.cpp")
  endif()
  add_library(${name} STATIC "${source}")
  configure(${name})
  target_compile_options(${name} PRIVATE -fstack-usage)
  add_executable(${name}_bench "${ROOT}/workstreams/inner_design/scheduling_20260919/benchmark.cpp")
  configure(${name}_bench)
  target_link_libraries(${name}_bench PRIVATE ${spin_target} ${name})
  add_executable(${name}_test "${BASE}/correctness.cpp")
  configure(${name}_test)
  target_link_libraries(${name}_test PRIVATE ${spin_target} ${name})
  add_test(NAME ${name} COMMAND ${name}_test)
  if(name STREQUAL control OR name STREQUAL dfs OR name STREQUAL four OR name MATCHES "^packed")
    add_executable(${name}_outer_bench outer_test.cpp)
    configure(${name}_outer_bench)
    target_link_libraries(${name}_outer_bench PRIVATE ${name})
    if(name STREQUAL four OR name MATCHES "^packed")
      target_compile_definitions(${name}_outer_bench PRIVATE BCH_FOUR=1)
    endif()
    if(name MATCHES "^packed")
      target_compile_definitions(${name}_outer_bench PRIVATE BCH_PACKED=1)
    endif()
    add_test(NAME ${name}_basis COMMAND ${name}_outer_bench)
  endif()
endforeach()
add_executable(packedshare4_tiles tile_test.cpp)
configure(packedshare4_tiles)
target_link_libraries(packedshare4_tiles PRIVATE spin_packedfour packedshare4)
add_test(NAME packedshare4_tiles COMMAND packedshare4_tiles)
