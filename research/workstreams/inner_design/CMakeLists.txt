cmake_minimum_required(VERSION 3.20)
project(ExactInnerBases LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BARE "${REPO_ROOT}/workstreams/bare_bch_rm2sub")
set(QUARTER "${REPO_ROOT}/workstreams/rate_quarter_bch/implementation")
set(VENDOR "${REPO_ROOT}/constructions/riffle_parityfanout31x33_bchperm_transpose_bitshuffle_splitstate_preaddmul_rm2sub_t128_s19/implementation/vendor")
set(SPIN_ARCH "znver4" CACHE STRING "GCC target architecture")
function(configure_kernel target)
    target_include_directories(${target} PUBLIC "${BARE}" "${QUARTER}/generated" "${VENDOR}")
    target_compile_definitions(${target} PRIVATE SPIN_GROUPED_A=1)
    if(MSVC)
        target_compile_options(${target} PRIVATE /arch:AVX2)
    else()
        target_compile_options(${target} PRIVATE "-march=${SPIN_ARCH}" -mavx2 -mpclmul -mvpclmulqdq)
    endif()
endfunction()
add_library(basis_outers STATIC "${BARE}/generated/BchCircuit.cpp" "${QUARTER}/generated/QuarterCircuit.cpp")
configure_kernel(basis_outers)
enable_testing()
file(GLOB candidate_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/generated/*/CandidateSpin.cpp")
foreach(source IN LISTS candidate_sources)
    get_filename_component(directory "${source}" DIRECTORY)
    get_filename_component(name "${directory}" NAME)
    add_library(basis_${name} STATIC "${source}")
    configure_kernel(basis_${name})
    target_link_libraries(basis_${name} PUBLIC basis_outers)
    add_executable(basis_${name}_bench "${BARE}/benchmark.cpp")
    configure_kernel(basis_${name}_bench)
    target_link_libraries(basis_${name}_bench PRIVATE basis_${name})
    add_executable(basis_${name}_test "${QUARTER}/correctness.cpp")
    configure_kernel(basis_${name}_test)
    target_link_libraries(basis_${name}_test PRIVATE basis_${name})
    add_test(NAME basis_${name}_test COMMAND basis_${name}_test)
endforeach()
