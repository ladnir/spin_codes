cmake_minimum_required(VERSION 3.20)
project(SpinCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(SPIN_ARCH native CACHE STRING "GCC/Clang target architecture; use znver4 on the paper's host")
if(MSVC OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    message(FATAL_ERROR "This package currently targets x86-64 with GCC or Clang")
endif()
add_library(spin_options INTERFACE)
target_compile_options(spin_options INTERFACE "-march=${SPIN_ARCH}" -mavx2 -mpclmul -mvpclmulqdq)
set(COMMON "${CMAKE_CURRENT_SOURCE_DIR}/src/common")
add_library(spin_outers STATIC "${COMMON}/generated/BchCircuit.cpp" "${COMMON}/generated/QuarterCircuit.cpp")
target_include_directories(spin_outers PUBLIC "${COMMON}" "${COMMON}/generated" "${CMAKE_CURRENT_SOURCE_DIR}/vendor")
target_link_libraries(spin_outers PUBLIC spin_options)

add_library(spin_half_transpose STATIC src/half/Weight5Spin.cpp)
target_link_libraries(spin_half_transpose PUBLIC spin_outers)
target_compile_definitions(spin_half_transpose PRIVATE SPIN_GROUPED_A=1 SPIN_WORKSPACE_ROUTING_OPT=1 W5_MASKED=0 W5_SHARED_EMISSION=0 W5_WRITE_PREFETCH=0 W5_PREFETCH=32)

add_library(spin_quarter_transpose STATIC src/routing/deployment/AsymmetricSpin.cpp)
target_link_libraries(spin_quarter_transpose PUBLIC spin_outers)
target_compile_definitions(spin_quarter_transpose PRIVATE SPIN_GROUPED_A=1 SPIN_WORKSPACE_ROUTING_OPT=1)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/bidirectional/Spin.cpp")
    add_library(spin_half_bidirectional STATIC src/bidirectional/Spin.cpp src/bidirectional/generated/BchCircuit.cpp src/bidirectional/generated/BchForward.cpp)
    target_include_directories(spin_half_bidirectional PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src/bidirectional")
    target_link_libraries(spin_half_bidirectional PUBLIC spin_options)
    target_compile_definitions(spin_half_bidirectional PRIVATE SPIN_GROUPED_A=1 SPIN_WORKSPACE_ROUTING_OPT=1)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(half_test tests/half.cpp)
    target_link_libraries(half_test PRIVATE spin_half_transpose)
    add_test(NAME half_transpose COMMAND half_test)
    add_executable(quarter_test tests/quarter.cpp)
    target_link_libraries(quarter_test PRIVATE spin_quarter_transpose)
    add_test(NAME quarter_transpose COMMAND quarter_test)
    add_executable(inner_test tests/inner.cpp)
    target_include_directories(inner_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/half")
    target_link_libraries(inner_test PRIVATE spin_outers)
    target_compile_definitions(inner_test PRIVATE SPIN_GROUPED_A=1 W5_SHARED_EMISSION=0)
    add_test(NAME half_inner_identity COMMAND inner_test)
    if(TARGET spin_half_bidirectional)
        add_executable(bidirectional_test tests/bidirectional.cpp)
        target_link_libraries(bidirectional_test PRIVATE spin_half_bidirectional)
        add_test(NAME half_bidirectional COMMAND bidirectional_test)
        set_tests_properties(half_bidirectional PROPERTIES RUN_SERIAL TRUE)
    endif()
    foreach(mode IN ITEMS off on failure portable)
        add_executable(page_${mode}_test tests/page.cpp)
        target_include_directories(page_${mode}_test PRIVATE "${COMMON}")
        if(NOT mode STREQUAL "off")
            target_compile_definitions(page_${mode}_test PRIVATE SPIN_WORKSPACE_ROUTING_OPT=1)
        endif()
        if(mode STREQUAL "failure")
            target_compile_definitions(page_${mode}_test PRIVATE SPIN_WORKSPACE_ROUTING_ADVICE_FAILURE_TEST=1)
        elseif(mode STREQUAL "portable")
            target_compile_definitions(page_${mode}_test PRIVATE SPIN_WORKSPACE_ROUTING_PORTABLE_TEST=1)
        endif()
        add_test(NAME page_${mode} COMMAND page_${mode}_test)
    endforeach()
    # Correctness suites are deliberately serialized, as are any user benchmarks.
    set_tests_properties(half_transpose quarter_transpose half_inner_identity page_off page_on page_failure page_portable PROPERTIES RUN_SERIAL TRUE)
endif()
