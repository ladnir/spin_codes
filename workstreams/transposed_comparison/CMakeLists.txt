cmake_minimum_required(VERSION 3.20)
project(SpinTransposedComparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(LIBOTE_SOURCE "" CACHE PATH "libOTe checkout at fa7bfc3c8a0178fc99b0b666d5f31fc2a9277579")
set(COMPARISON_ARCH "znver4" CACHE STRING "Benchmark CPU architecture")
if(NOT EXISTS "${LIBOTE_SOURCE}/libOTe/Tools/Ppcg/ChosenBlockBaa.h")
  message(FATAL_ERROR "Set LIBOTE_SOURCE to the pinned chosen-block-baa checkout")
endif()
set(ENABLE_BOOST OFF CACHE BOOL "" FORCE)
set(ENABLE_SODIUM OFF CACHE BOOL "" FORCE)
set(ENABLE_RELIC OFF CACHE BOOL "" FORCE)
set(FETCH_AUTO ON CACHE BOOL "" FORCE)
if(NOT MSVC)
  add_compile_options(-O3 -DNDEBUG "-march=${COMPARISON_ARCH}" -mavx2 -mpclmul -mvpclmulqdq)
else()
  add_compile_options(/O2 /arch:AVX2)
endif()
file(MAKE_DIRECTORY "${LIBOTE_SOURCE}/cryptoTools/out")
add_subdirectory("${LIBOTE_SOURCE}/cryptoTools" cryptoTools EXCLUDE_FROM_ALL)
# Only encoder headers are used; no OT/VOLE protocols are enabled.
configure_file("${LIBOTE_SOURCE}/libOTe/config.h.in" "libOTe/config.h")
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(transposed_benchmark benchmark.cpp
  "${LIBOTE_SOURCE}/libOTe/Tools/ExConvCodeOld/ExConvCodeInstantiations.cpp"
  "${REPO_ROOT}/workstreams/bare_bch_rm2sub/Spin.cpp"
  "${REPO_ROOT}/workstreams/bare_bch_rm2sub/generated/BchCircuit.cpp")
target_include_directories(transposed_benchmark PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${LIBOTE_SOURCE}"
  "${REPO_ROOT}/workstreams/bare_bch_rm2sub")
target_compile_definitions(transposed_benchmark PRIVATE SPIN_GROUPED_A=1 LIBOTE_ENABLE_OLD_EXCONV=1)
target_link_libraries(transposed_benchmark PRIVATE cryptoTools)
enable_testing()
add_test(NAME comparison_correctness COMMAND transposed_benchmark check)
