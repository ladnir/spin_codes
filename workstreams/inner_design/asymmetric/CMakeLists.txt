cmake_minimum_required(VERSION 3.20)
project(AsymmetricInnerDiagnostics LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(VENDOR "${ROOT}/constructions/riffle_parityfanout31x33_bchperm_transpose_bitshuffle_splitstate_preaddmul_rm2sub_t128_s19/implementation/vendor")
foreach(variant IN ITEMS balanced greedy3_2 mixed2_3)
    set(target inner_${variant}_bench)
    add_executable(${target} kernel_benchmark.cpp)
    if(variant STREQUAL "balanced")
        set(directory balanced_masked)
        target_compile_definitions(${target} PRIVATE ASYM_BASELINE=1)
    else()
        set(directory asymmetric_${variant}_sparse)
        target_compile_definitions(${target} PRIVATE ASYM_BASELINE=0)
    endif()
    target_include_directories(${target} PRIVATE "${ROOT}/workstreams/bare_bch_rm2sub" "${VENDOR}" "../generated/${directory}")
    target_compile_definitions(${target} PRIVATE SPIN_GROUPED_A=1 ASYM_NAME="${variant}")
    target_compile_options(${target} PRIVATE -march=znver4 -mavx2 -mpclmul -mvpclmulqdq)
endforeach()
