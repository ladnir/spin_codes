cmake_minimum_required(VERSION 3.20)
project(SpinRoutingDeployment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(BARE "${ROOT}/workstreams/bare_bch_rm2sub")
set(QUARTER "${ROOT}/workstreams/rate_quarter_bch/implementation")
set(VENDOR "${ROOT}/constructions/riffle_parityfanout31x33_bchperm_transpose_bitshuffle_splitstate_preaddmul_rm2sub_t128_s19/implementation/vendor")
set(SPIN_ARCH znver4 CACHE STRING "GCC architecture")
function(configure target)
    target_include_directories(${target} PRIVATE "${BARE}" "${QUARTER}/generated" "${VENDOR}")
    if(MSVC)
        target_compile_options(${target} PRIVATE /arch:AVX2)
    else()
        target_compile_options(${target} PRIVATE "-march=${SPIN_ARCH}" -mavx2 -mpclmul -mvpclmulqdq)
    endif()
endfunction()
add_library(outers STATIC "${BARE}/generated/BchCircuit.cpp" "${QUARTER}/generated/QuarterCircuit.cpp")
configure(outers)
enable_testing()
foreach(name supported_off supported_on asymmetric_off asymmetric_on asymmetric_fallback asymmetric_portable)
    if(name MATCHES "^supported")
        set(source "${BARE}/Spin.cpp")
    else()
        set(source "${CMAKE_CURRENT_SOURCE_DIR}/AsymmetricSpin.cpp")
    endif()
    if(name MATCHES "_off$")
        set(enabled 0)
    else()
        set(enabled 1)
    endif()
    add_library(${name} STATIC "${source}")
    configure(${name})
    target_compile_definitions(${name} PRIVATE SPIN_GROUPED_A=1 SPIN_WORKSPACE_ROUTING_OPT=${enabled})
    if(name MATCHES "_fallback$")
        target_compile_definitions(${name} PRIVATE SPIN_WORKSPACE_ROUTING_ADVICE_FAILURE_TEST=1)
    elseif(name MATCHES "_portable$")
        target_compile_definitions(${name} PRIVATE SPIN_WORKSPACE_ROUTING_PORTABLE_TEST=1)
    endif()
    target_link_libraries(${name} PUBLIC outers)
    add_executable(${name}_test "${QUARTER}/correctness.cpp")
    configure(${name}_test)
    target_link_libraries(${name}_test PRIVATE ${name})
    add_test(NAME ${name}_test COMMAND ${name}_test)
    if(name MATCHES "(_off|_on)$")
        add_executable(${name}_bench "${BARE}/benchmark.cpp")
        configure(${name}_bench)
        target_link_libraries(${name}_bench PRIVATE ${name})
    endif()
    if(name MATCHES "^supported")
        add_executable(${name}_half_test "${BARE}/correctness.cpp")
        configure(${name}_half_test)
        target_link_libraries(${name}_half_test PRIVATE ${name})
        add_test(NAME ${name}_half_test COMMAND ${name}_half_test)
    endif()
endforeach()
foreach(mode off on failure portable)
    add_executable(page_${mode}_test page_test.cpp)
    target_include_directories(page_${mode}_test PRIVATE "${BARE}")
    if(NOT mode STREQUAL "off")
        target_compile_definitions(page_${mode}_test PRIVATE SPIN_WORKSPACE_ROUTING_OPT=1)
    endif()
    if(mode STREQUAL "failure")
        target_compile_definitions(page_${mode}_test PRIVATE SPIN_WORKSPACE_ROUTING_ADVICE_FAILURE_TEST=1)
    elseif(mode STREQUAL "portable")
        target_compile_definitions(page_${mode}_test PRIVATE SPIN_WORKSPACE_ROUTING_PORTABLE_TEST=1)
    endif()
    add_test(NAME page_${mode}_test COMMAND page_${mode}_test)
endforeach()
